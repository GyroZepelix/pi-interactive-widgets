function info(b64: string) {
  const b = Buffer.from(b64, "base64"); let o = 8; const idat: Buffer[] = []; let ct = -1, w = 0, h = 0;
  while (o < b.length) { const len = b.readUInt32BE(o); const type = b.toString("latin1", o + 4, o + 8); const d = b.subarray(o + 8, o + 8 + len);
    if (type === "IHDR") { w = d.readUInt32BE(0); h = d.readUInt32BE(4); ct = d[9]; } if (type === "IDAT") idat.push(d); o += 12 + len; }
  let corner: number[] = [];
  if (ct === 6) { const raw = inflateSync(Buffer.concat(idat)); corner = [...raw.subarray(1, 5)]; }
  return { colorType: ct, hasAlpha: ct === 6 || ct === 4, w, h, firstPixelRGBA: corner };
}
